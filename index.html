1. 구글 시트에서 [확장 프로그램 > Apps Script]를 열고 기존 코드를 지운 뒤, 위 버튼으로 복사한 코드를 붙여넣고 저장(Ctrl+S)합니다.

2. 구글 시트 내 종목코드 시트의 E열(종목코드)과 F열(종목명) 데이터를 자동으로 읽어와 종목번호 및 종목명을 즉시 찾아줍니다.

3. 우측 상단 [배포 > 새 배포] (또는 [배포 > 배포 관리 > ✏️ 수정 > 버전: 새 버전]) → 유형: 웹 앱 → 액세스 권한: 모든 사용자로 배포한 뒤 웹 앱 URL을 아래에 연결하세요.

var TRADES_SHEET_NAME = '거래내역';
var LEGACY_TRADES_SHEET_NAME = '매매내역';
var PORTFOLIO_SHEET_NAME = '포트폴리오';
var STOCK_CODE_SHEET_NAME = '종목코드';

function normalizeStockCodeStr(raw) {
  if (!raw) return '';
  var code = String(raw).trim().toUpperCase().replace(/^(KRX:|KOSDAQ:|KOSPI:|NASDAQ:|NYSE:|AMEX:)/, '');
  if (/^\d{1,5}$/.test(code)) code = ('000000' + code).slice(-6);
  return code;
}

function toGfTicker(raw) {
  if (!raw) return '';
  var code = String(raw).trim().toUpperCase();
  if (/^\d{1,5}$/.test(code)) code = 'KRX:' + ('000000' + code).slice(-6);
  else if (/^\d{6}$/.test(code)) code = 'KRX:' + code;
  else if (/^\d{4}[0-9A-Z]{2}$/.test(code) && code.indexOf(':') === -1) code = 'KRX:' + code;
  return code;
}

function getOrCreateSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var tradesSheet = ss.getSheetByName(TRADES_SHEET_NAME);
  if (!tradesSheet) {
    var legacy = ss.getSheetByName(LEGACY_TRADES_SHEET_NAME);
    if (legacy) {
      legacy.setName(TRADES_SHEET_NAME);
      tradesSheet = legacy;
    } else {
      tradesSheet = ss.insertSheet(TRADES_SHEET_NAME);
    }
  }
  if (tradesSheet.getLastRow() === 0) {
    tradesSheet.appendRow(['id', 'date', 'type', 'currency', 'sector', 'code', 'name', 'price', 'quantity', 'createdAt']);
  }

  var pfSheet = ss.getSheetByName(PORTFOLIO_SHEET_NAME);
  if (!pfSheet) {
    pfSheet = ss.insertSheet(PORTFOLIO_SHEET_NAME);
  }
  if (pfSheet.getLastRow() === 0) {
    pfSheet.appendRow(['code', 'name', 'currency', 'sector', 'avgPrice', 'currentPrice', 'quantity', 'lastDate', 'manualCurrentPrice']);
  }
  return { ss: ss, tradesSheet: tradesSheet, pfSheet: pfSheet };
}

function normalizeIsoDate(val) {
  if (val instanceof Date) {
    return Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd');
  }
  var s = String(val || '').trim().replace(/^'+/, '');
  var m4 = s.match(/^(\d{4})[-./](\d{1,2})[-./](\d{1,2})/);
  if (m4) {
    return m4[1] + '-' + ('0' + m4[2]).slice(-2) + '-' + ('0' + m4[3]).slice(-2);
  }
  var m2 = s.match(/^(\d{2})[-./](\d{1,2})[-./](\d{1,2})/);
  if (m2) {
    return '20' + m2[1] + '-' + ('0' + m2[2]).slice(-2) + '-' + ('0' + m2[3]).slice(-2);
  }
  return s.slice(0, 10);
}

function toShortDateDisplay(isoStr) {
  var norm = normalizeIsoDate(isoStr);
  var m = norm.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (m) {
    return "'" + m[1].slice(2) + '/' + m[2] + '/' + m[3];
  }
  return norm;
}

// [종목코드] 시트의 E열(종목코드)과 F열(종목명) 전체 읽기
function readStockCodeSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var codeSheet = ss.getSheetByName(STOCK_CODE_SHEET_NAME);
  if (!codeSheet) return [];
  var lastRow = codeSheet.getLastRow();
  if (lastRow < 1) return [];

  // E열(5번째 열) = 종목코드, F열(6번째 열) = 종목명
  var vals = codeSheet.getRange(1, 5, lastRow, 2).getDisplayValues();
  var list = [];
  for (var i = 0; i < vals.length; i++) {
    var rawCode = String(vals[i][0] || '').trim();
    var rawName = String(vals[i][1] || '').trim();
    if (!rawCode || !rawName) continue;
    // 헤더 행 제외
    var upperCode = rawCode.toUpperCase();
    if (upperCode === '종목코드' || upperCode === '종목번호' || upperCode === '코드' || upperCode === 'CODE' || upperCode === 'TICKER' || rawName === '종목명') {
      continue;
    }
    var cleanCode = normalizeStockCodeStr(rawCode);
    list.push([cleanCode, rawName]);
  }
  return list;
}

// [종목코드] 시트의 E열(종목코드)과 F열(종목명)을 참조하여 검색
function searchStockApi(query) {
  var q = String(query || '').trim().toUpperCase().replace(/\s+/g, '');
  if (!q) return [];
  var rows = readStockCodeSheet();
  var exact = [];
  var prefix = [];
  var partial = [];

  for (var i = 0; i < rows.length; i++) {
    var code = rows[i][0];
    var name = rows[i][1];
    var cUp = code.toUpperCase();
    var nUp = name.toUpperCase().replace(/\s+/g, '');
    var isKr = /^\d{6}$/.test(cUp) || /^\d{4}[0-9A-Z]{2}$/.test(cUp);
    var item = {
      code: code,
      name: name,
      exchange: '종목코드',
      currency: isKr ? 'KRW' : 'USD'
    };

    if (cUp === q || nUp === q) {
      exact.push(item);
    } else if (cUp.indexOf(q) === 0 || nUp.indexOf(q) === 0) {
      prefix.push(item);
    } else if (cUp.indexOf(q) !== -1 || nUp.indexOf(q) !== -1) {
      partial.push(item);
    }
  }

  return exact.concat(prefix).concat(partial);
}

// 거래내역 시트를 시간의 역순 정렬 및 포트폴리오 시트(보유수량 > 0, 평균단가 자동계산) 갱신
function rebuildPortfolioSheet(tradesSheet, pfSheet, overrideMap) {
  var lastRow = tradesSheet.getLastRow();
  var trades = [];
  if (lastRow > 1) {
    var vals = tradesSheet.getRange(2, 1, lastRow - 1, 10).getValues();
    for (var i = 0; i < vals.length; i++) {
      var r = vals[i];
      if (!r[0]) continue;
      trades.push({
        id: String(r[0]),
        date: normalizeIsoDate(r[1]),
        type: String(r[2] || 'BUY').toUpperCase(),
        currency: String(r[3] || 'KRW'),
        sector: String(r[4] || ''),
        code: toGfTicker(r[5]),
        name: String(r[6] || ''),
        price: Number(r[7]) || 0,
        quantity: Number(r[8]) || 0,
        createdAt: Number(r[9]) || 0
      });
    }
  }

  trades.sort(function(a, b) {
    if (a.date !== b.date) return a.date < b.date ? 1 : -1;
    return (b.createdAt || 0) - (a.createdAt || 0);
  });

  if (lastRow > 1) {
    tradesSheet.getRange(2, 1, lastRow - 1, 10).clearContent();
  }
  if (trades.length > 0) {
    var tradeRows = trades.map(function(t) {
      return [t.id, "'" + toShortDateDisplay(t.date), t.type, t.currency, t.sector, t.code, t.name, t.price, t.quantity, t.createdAt];
    });
    tradesSheet.getRange(2, 1, tradeRows.length, 10).setValues(tradeRows);
  }

  var existingOverrides = overrideMap || {};
  var pfLastRow = pfSheet.getLastRow();
  if (pfLastRow > 1) {
    var pfVals = pfSheet.getRange(2, 1, pfLastRow - 1, 9).getValues();
    for (var p = 0; p < pfVals.length; p++) {
      var key = String(pfVals[p][0] || pfVals[p][1]).trim().toUpperCase();
      if (!key) continue;
      if (!existingOverrides[key]) existingOverrides[key] = {};
      if (pfVals[p][8] !== '' && pfVals[p][8] !== null && !isNaN(Number(pfVals[p][8])) && Number(pfVals[p][8]) > 0) {
        if (existingOverrides[key].manualCurrentPrice === undefined) {
          existingOverrides[key].manualCurrentPrice = Number(pfVals[p][8]);
        }
      }
    }
  }

  var chrono = trades.slice().reverse();
  var map = {};
  for (var c = 0; c < chrono.length; c++) {
    var tr = chrono[c];
    var tickerKey = (tr.code || tr.name).trim().toUpperCase();
    if (!tickerKey) continue;
    if (!map[tickerKey]) {
      map[tickerKey] = {
        key: tickerKey,
        code: tr.code,
        name: tr.name,
        currency: tr.currency || 'KRW',
        sector: tr.sector || '',
        avgBuyPrice: 0,
        quantity: 0,
        lastDate: tr.date
      };
    }
    var entry = map[tickerKey];
    if (tr.name) entry.name = tr.name;
    if (tr.sector) entry.sector = tr.sector;
    if (tr.currency) entry.currency = tr.currency;
    if (tr.code) entry.code = tr.code;
    if (tr.date >= entry.lastDate) entry.lastDate = tr.date;

    if (tr.type === 'SELL' || tr.type === '매도') {
      entry.quantity -= tr.quantity;
      if (entry.quantity <= 0) {
        entry.quantity = 0;
        entry.avgBuyPrice = 0;
      }
    } else {
      var prevCost = entry.quantity * entry.avgBuyPrice;
      var newCost = tr.quantity * tr.price;
      entry.quantity += tr.quantity;
      entry.avgBuyPrice = entry.quantity > 0 ? (prevCost + newCost) / entry.quantity : tr.price;
    }
  }

  var activeHoldings = Object.keys(map)
    .map(function(k) { return map[k]; })
    .filter(function(item) { return item.quantity > 0.000001; });

  if (pfLastRow > 1) {
    pfSheet.getRange(2, 1, pfLastRow - 1, 9).clearContent();
  }

  for (var h = 0; h < activeHoldings.length; h++) {
    var hold = activeHoldings[h];
    var rowNum = h + 2;
    var ov = existingOverrides[hold.key] || {};
    var manualCur = (ov.manualCurrentPrice !== undefined && ov.manualCurrentPrice !== null && ov.manualCurrentPrice > 0)
      ? Number(ov.manualCurrentPrice)
      : '';

    pfSheet.getRange(rowNum, 1, 1, 9).setValues([[
      hold.code,
      hold.name,
      hold.currency,
      hold.sector,
      hold.avgBuyPrice,
      manualCur !== '' ? manualCur : hold.avgBuyPrice,
      hold.quantity,
      hold.lastDate,
      ov.manualCurrentPrice || ''
    ]]);

    if (hold.code && manualCur === '') {
      pfSheet.getRange(rowNum, 6).setFormula('=IFERROR(GOOGLEFINANCE("' + hold.code + '", "price"), E' + rowNum + ')');
    }
  }

  SpreadsheetApp.flush();
}

function readState(tradesSheet, pfSheet) {
  SpreadsheetApp.flush();
  var trades = [];
  var tLast = tradesSheet.getLastRow();
  if (tLast > 1) {
    var tVals = tradesSheet.getRange(2, 1, tLast - 1, 10).getValues();
    for (var i = 0; i < tVals.length; i++) {
      var r = tVals[i];
      if (!r[0]) continue;
      trades.push({
        id: String(r[0]),
        date: normalizeIsoDate(r[1]),
        type: String(r[2] || 'BUY').toUpperCase(),
        currency: String(r[3] || 'KRW'),
        sector: String(r[4] || ''),
        code: String(r[5] || ''),
        name: String(r[6] || ''),
        price: Number(r[7]) || 0,
        quantity: Number(r[8]) || 0,
        createdAt: Number(r[9]) || 0
      });
    }
  }

  var items = [];
  var pLast = pfSheet.getLastRow();
  if (pLast > 1) {
    var pVals = pfSheet.getRange(2, 1, pLast - 1, 9).getValues();
    for (var j = 0; j < pVals.length; j++) {
      var p = pVals[j];
      var code = String(p[0] || '');
      var name = String(p[1] || '');
      var qty = Number(p[6]) || 0;
      if ((!code && !name) || qty <= 0) continue;
      var key = (code || name).trim().toUpperCase();
      var avgPrice = Number(p[4]) || 0;
      var curPrice = Number(p[5]) || avgPrice;
      items.push({
        id: key,
        code: code,
        name: name,
        currency: String(p[2] || 'KRW'),
        sector: String(p[3] || ''),
        price: avgPrice,
        currentPrice: curPrice,
        quantity: qty,
        date: normalizeIsoDate(p[7]),
        manualCurrentPrice: p[8] !== '' ? Number(p[8]) : null
      });
    }
  }
  return { ok: true, trades: trades, items: items, stockCodes: readStockCodeSheet() };
}

function doGet(e) {
  var searchQuery = (e && e.parameter && e.parameter.search) ? e.parameter.search.trim() : '';
  if (searchQuery) {
    var results = searchStockApi(searchQuery);
    return ContentService.createTextOutput(JSON.stringify({ ok: true, results: results }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var sheets = getOrCreateSheets();
  var ticker = (e && e.parameter && e.parameter.ticker) ? e.parameter.ticker.trim() : '';
  if (ticker) {
    var gfTicker = toGfTicker(ticker);
    var cell = sheets.pfSheet.getRange('Z1');
    cell.setFormula('=IFERROR(GOOGLEFINANCE("' + gfTicker + '", "price"), "")');
    SpreadsheetApp.flush();
    var val = Number(cell.getValue());
    cell.clearContent();
    return ContentService.createTextOutput(JSON.stringify({ ok: true, ticker: gfTicker, price: val }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  return ContentService.createTextOutput(JSON.stringify(readState(sheets.tradesSheet, sheets.pfSheet)))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheets = getOrCreateSheets();
    var tradesSheet = sheets.tradesSheet;
    var pfSheet = sheets.pfSheet;
    var body = JSON.parse(e.postData.contents);
    var action = body.action;

    if (action === 'addTrade') {
      var tr = body.trade;
      var gfTicker = toGfTicker(tr.code);
      tradesSheet.appendRow([
        String(tr.id),
        normalizeIsoDate(tr.date),
        String(tr.type || 'BUY').toUpperCase(),
        String(tr.currency || 'KRW'),
        String(tr.sector || ''),
        gfTicker || String(tr.code || ''),
        String(tr.name || ''),
        Number(tr.price) || 0,
        Number(tr.quantity) || 0,
        Number(tr.createdAt) || Date.now()
      ]);
      var ovMap = {};
      var key = (gfTicker || tr.name || '').trim().toUpperCase();
      if (key && tr.manualCurrentPrice && Number(tr.currentPrice) > 0) {
        ovMap[key] = { manualCurrentPrice: Number(tr.currentPrice) };
      }
      rebuildPortfolioSheet(tradesSheet, pfSheet, ovMap);
    } else if (action === 'updateTrade') {
      var targetTradeId = String(body.id);
      var tLastRow = tradesSheet.getLastRow();
      if (tLastRow > 1) {
        var tData = tradesSheet.getRange(2, 1, tLastRow - 1, 9).getValues();
        for (var u = 0; u < tData.length; u++) {
          if (String(tData[u][0]) === targetTradeId) {
            if (body.type !== undefined) tradesSheet.getRange(u + 2, 3).setValue(String(body.type).toUpperCase());
            if (body.currency !== undefined) tradesSheet.getRange(u + 2, 4).setValue(String(body.currency).toUpperCase());
            if (body.price !== undefined) tradesSheet.getRange(u + 2, 8).setValue(Number(body.price));
            if (body.quantity !== undefined) tradesSheet.getRange(u + 2, 9).setValue(Number(body.quantity));
            break;
          }
        }
      }
      rebuildPortfolioSheet(tradesSheet, pfSheet);
    } else if (action === 'deleteTrade' || action === 'deleteTrades') {
      var delIds = Array.isArray(body.ids) ? body.ids.map(String) : [String(body.id)];
      var delSet = {};
      for (var d = 0; d < delIds.length; d++) delSet[delIds[d]] = true;
      var tLast = tradesSheet.getLastRow();
      if (tLast > 1) {
        var tVals = tradesSheet.getRange(2, 1, tLast - 1, 1).getValues();
        for (var i = tVals.length - 1; i >= 0; i--) {
          if (delSet[String(tVals[i][0])]) {
            tradesSheet.deleteRow(i + 2);
          }
        }
      }
      rebuildPortfolioSheet(tradesSheet, pfSheet);
    } else if (action === 'updatePortfolio') {
      var targetKey = String(body.id || '').trim().toUpperCase();
      var ov = {};
      ov[targetKey] = {};
      if (body.currentPrice !== undefined) ov[targetKey].manualCurrentPrice = Number(body.currentPrice);
      rebuildPortfolioSheet(tradesSheet, pfSheet, ov);
    } else if (action === 'syncAll' && Array.isArray(body.trades)) {
      var existingLast = tradesSheet.getLastRow();
      var idSet = {};
      if (existingLast > 1) {
        var exIds = tradesSheet.getRange(2, 1, existingLast - 1, 1).getValues();
        for (var k = 0; k < exIds.length; k++) {
          idSet[String(exIds[k][0])] = true;
        }
      }
      body.trades.forEach(function(tr) {
        if (!idSet[String(tr.id)]) {
          var tk = toGfTicker(tr.code);
          tradesSheet.appendRow([
            String(tr.id),
            normalizeIsoDate(tr.date),
            String(tr.type || 'BUY').toUpperCase(),
            String(tr.currency || 'KRW'),
            String(tr.sector || ''),
            tk || String(tr.code || ''),
            String(tr.name || ''),
            Number(tr.price) || 0,
            Number(tr.quantity) || 0,
            Number(tr.createdAt) || Date.now()
          ]);
        }
      });
      rebuildPortfolioSheet(tradesSheet, pfSheet);
    }

    return ContentService.createTextOutput(JSON.stringify(readState(tradesSheet, pfSheet)))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
