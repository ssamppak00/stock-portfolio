1. 새 구글 시트(https://sheets.new)를 열고 상단 메뉴 [확장 프로그램 > Apps Script]를 클릭합니다.

2. 기존 코드를 모두 지우고 위 [📋 구글 시트 Apps Script 코드 복사하기] 버튼으로 복사한 코드를 붙여넣은 뒤 저장(Ctrl+S)합니다.

3. 우측 상단 [배포 > 새 배포] 클릭 → 유형 선택(⚙️): 웹 앱 → 다음 사용자로 실행: 나 → 액세스 권한이 있는 사용자: 모든 사용자로 선택 후 [배포]합니다.

4. 발급된 웹 앱 URL (https://script.google.com/macros/s/.../exec)을 아래 입력칸에 붙여넣고 [연결 저장]을 누르면 모든 기기에서 동일한 포트폴리오가 공유됩니다.

function getOrInitSheet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['id', 'date', 'currency', 'sector', 'code', 'name', 'buyPrice', 'currentPrice', 'quantity']);
  }
  return sheet;
}

function toGfTicker(raw) {
  if (!raw) return '';
  var code = String(raw).trim().toUpperCase();
  if (/^\d{1,5}$/.test(code)) code = 'KRX:' + ('000000' + code).slice(-6);
  else if (/^\d{6}$/.test(code)) code = 'KRX:' + code;
  else if (/^\d{4}[0-9A-Z]{2}$/.test(code) && code.indexOf(':') === -1) code = 'KRX:' + code;
  return code;
}

function doGet(e) {
  var sheet = getOrInitSheet();
  var ticker = (e && e.parameter && e.parameter.ticker) ? e.parameter.ticker.trim() : '';
  if (ticker) {
    var gfTicker = toGfTicker(ticker);
    var cell = sheet.getRange('Z1');
    cell.setFormula('=IFERROR(GOOGLEFINANCE("' + gfTicker + '", "price"), "")');
    SpreadsheetApp.flush();
    var val = Number(cell.getValue());
    cell.clearContent();
    return ContentService.createTextOutput(JSON.stringify({ ok: true, ticker: gfTicker, price: val }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  SpreadsheetApp.flush();
  var values = sheet.getDataRange().getValues();
  var items = [];
  for (var i = 1; i < values.length; i++) {
    var r = values[i];
    if (!r[0]) continue;
    var rawDate = r[1];
    var dateStr = (rawDate instanceof Date)
      ? Utilities.formatDate(rawDate, Session.getScriptTimeZone(), 'yyyy-MM-dd')
      : String(rawDate).slice(0, 10);
    var buyPrice = Number(r[6]) || 0;
    var curPrice = Number(r[7]) || buyPrice;
    items.push({
      id: String(r[0]),
      date: dateStr,
      currency: String(r[2] || 'KRW'),
      sector: String(r[3] || ''),
      code: String(r[4] || ''),
      name: String(r[5] || ''),
      price: buyPrice,
      currentPrice: curPrice,
      quantity: Number(r[8]) || 0
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, items: items }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var sheet = getOrInitSheet();
    var body = JSON.parse(e.postData.contents);
    var action = body.action;

    if (action === 'add') {
      var it = body.item;
      var rowNum = sheet.getLastRow() + 1;
      var gfTicker = toGfTicker(it.code);
      sheet.appendRow([
        String(it.id),
        String(it.date),
        String(it.currency || 'KRW'),
        String(it.sector || ''),
        gfTicker || String(it.code || ''),
        String(it.name || ''),
        Number(it.price) || 0,
        Number(it.currentPrice) || Number(it.price) || 0,
        Number(it.quantity) || 0
      ]);
      if (gfTicker && !it.manualCurrentPrice) {
        sheet.getRange(rowNum, 8).setFormula('=IFERROR(GOOGLEFINANCE("' + gfTicker + '", "price"), G' + rowNum + ')');
      }
      SpreadsheetApp.flush();
    } else if (action === 'update') {
      var targetId = String(body.id);
      var rows = sheet.getDataRange().getValues();
      for (var i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === targetId) {
          var rIdx = i + 1;
          if (body.price !== undefined) sheet.getRange(rIdx, 7).setValue(Number(body.price));
          if (body.currentPrice !== undefined) sheet.getRange(rIdx, 8).setValue(Number(body.currentPrice));
          if (body.restoreFormula && rows[i][4]) {
            var tk = toGfTicker(rows[i][4]);
            sheet.getRange(rIdx, 8).setFormula('=IFERROR(GOOGLEFINANCE("' + tk + '", "price"), G' + rIdx + ')');
          }
          if (body.quantity !== undefined) sheet.getRange(rIdx, 9).setValue(Number(body.quantity));
          break;
        }
      }
      SpreadsheetApp.flush();
    } else if (action === 'delete') {
      var delId = String(body.id);
      var allRows = sheet.getDataRange().getValues();
      for (var j = allRows.length - 1; j >= 1; j--) {
        if (String(allRows[j][0]) === delId) {
          sheet.deleteRow(j + 1);
          break;
        }
      }
      SpreadsheetApp.flush();
    } else if (action === 'syncAll' && Array.isArray(body.items)) {
      var existing = sheet.getDataRange().getValues();
      var idSet = {};
      for (var k = 1; k < existing.length; k++) {
        idSet[String(existing[k][0])] = true;
      }
      body.items.forEach(function(item) {
        if (!idSet[String(item.id)]) {
          var nextRow = sheet.getLastRow() + 1;
          var tk = toGfTicker(item.code);
          sheet.appendRow([
            String(item.id),
            String(item.date),
            String(item.currency || 'KRW'),
            String(item.sector || ''),
            tk || String(item.code || ''),
            String(item.name || ''),
            Number(item.price) || 0,
            Number(item.currentPrice) || Number(item.price) || 0,
            Number(item.quantity) || 0
          ]);
          if (tk) {
            sheet.getRange(nextRow, 8).setFormula('=IFERROR(GOOGLEFINANCE("' + tk + '", "price"), G' + nextRow + ')');
          }
        }
      });
      SpreadsheetApp.flush();
    }

    return ContentService.createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
